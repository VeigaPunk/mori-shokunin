grade_lane() {
  local lane=$1 repo
  repo=$(lane_repo "$lane")
  GL_STATE=''; GL_ROUND='-'; GL_LAUNCH='-'; GL_AGE='-'; GL_BASIS=''; GL_EXTRA=''
  GL_BRANCH=''; GL_HBM=0; GL_HEAD_SHORT=''; GL_BRANCH_TIP=''; GL_AMN=0

  # -- row 1: unreadable inputs ------------------------------------------------
  if [ ! -r "$TIMELINE_FILE" ]; then
    GL_STATE=unavailable; GL_BASIS='row-1'
    GL_EXTRA="reason=ledger-unreadable path=$TIMELINE_FILE"
    return 0
  fi
  if [ ! -d "$repo" ]; then
    GL_STATE=unavailable; GL_BASIS='row-1'
    GL_EXTRA="reason=repo-absent repo=$repo"
    return 0
  fi
  git_cap "$repo" rev-parse --git-dir
  if [ "$CAP_RC" -ne 0 ]; then
    GL_STATE=unavailable; GL_BASIS='row-1'
    GL_EXTRA="reason=git-unreadable rc=$CAP_RC repo=$repo"
    return 0
  fi
  ps_walk "$lane" "$repo"
  if [ "$PS_WALK_RC" -ne 0 ]; then
    GL_STATE=unavailable; GL_BASIS='row-1'
    GL_EXTRA="reason=tree-unreadable rc=$PS_WALK_RC repo=$repo"
    return 0
  fi

  # -- r17 C2 branch anchor ---------------------------------------------------
  # Resolve the branch once, and measure the HEAD-vs-branch invariant HERE,
  # before every early return, so a jj-drifted HEAD is reported for every lane
  # this census can read — not only for lanes that happen to grade through rows
  # A/B/C. Note the split (A2): a non-jj repo without a resolvable branch keeps
  # the entering HEAD anchor, while a `.jj`-colocated repo without an
  # authoritative lane.branch mirror is UNMEASURABLE here (reason=ref-unresolved)
  # and is never graded from an attached side bookmark.
  if lane_branch_probe "$repo"; then
    GL_BRANCH=$BRANCH_OUT
  elif [ -d "$repo/.jj" ]; then
    GL_STATE=unavailable; GL_BASIS='row-1'
    GL_EXTRA="reason=ref-unresolved repo=$repo"
    return 0
  fi
  if [ -n "$GL_BRANCH" ]; then head_branch_probe "$repo" "$GL_BRANCH"; fi

  # -- rows 2-8: launch anchor, closure, age ----------------------------------
  ledger_anchor "$lane"
  # r17 C2 / CensusNewBranches: the committer-author diagnostic is a window
  # observation, not a row grade. It runs HERE, before the young/report-closed
  # early returns below, exactly like the HEAD diagnostic runs before them — a
  # young lane (age <= gate) or a lane with an already committed report must not
  # hide the alert. The same strict (launch, NOW] window is applied inside.
  if [ "$LA_LINE" -gt 0 ] && is_int "$LA_EPOCH" && [ "$LA_EPOCH" -le "$NOW_EPOCH" ]; then
    lane_author_probe "$lane" "$repo" "$LA_EPOCH"
  fi
  if [ "$LA_LINE" -eq 0 ]; then
    GL_STATE=not-open; GL_BASIS='row-2'; GL_EXTRA='reason=no-launch-line'
    return 0
  fi
  if ! is_int "$LA_EPOCH"; then
    GL_STATE=unavailable; GL_BASIS='row-1'; GL_EXTRA='reason=launch-ts-unparseable'
    return 0
  fi
  GL_ROUND=$LA_ROUND; GL_LAUNCH=$LA_EPOCH
  if [ "$LA_EPOCH" -gt "$NOW_EPOCH" ]; then
    GL_STATE=young; GL_BASIS='row-3'; GL_EXTRA='reason=launch-in-future'
    return 0
  fi
  if [ "$LA_CLOSED" -eq 1 ]; then
    GL_STATE=not-open; GL_BASIS='row-4'
    GL_EXTRA="reason=ledger-close round=$LA_ROUND"
    return 0
  fi
  local round_state
  round_state=$(read_file "$STATE_DIR/$lane.round")
  if is_int "$round_state" && is_int "$LA_ROUND" && [ "$round_state" -gt "$LA_ROUND" ]; then
    GL_STATE=not-open; GL_BASIS='row-7'
    GL_EXTRA="reason=round-advanced state_round=$round_state anchor_round=$LA_ROUND"
    return 0
  fi
  local age=$((NOW_EPOCH - LA_EPOCH))
  GL_AGE=$age
  if [ "$age" -le "$LANE_SILENCE_S" ]; then
    GL_STATE=young; GL_BASIS='row-8'; GL_EXTRA="reason=age_le_gate gate_s=$LANE_SILENCE_S"
    return 0
  fi

  # -- facts shared by rows 6 and 9 --------------------------------------------
  # Every git fact below is read from refs/heads/<branch> when one resolves.
  # A non-jj repo with no resolvable branch keeps the entering HEAD anchor
  # (r13/r14/r15 stubs answer empty/rc0 for symbolic-ref, and their output must
  # not change); a colocated repo never reaches this point unresolved.
  # Safety (audit-r11 r13 advisory): `ls-files` can execute the repo's
  # core.fsmonitor command, and an inherited GIT_DIR/GIT_WORK_TREE/GIT_INDEX_FILE
  # overrides `git -C`. This gate is read-only and must query the named repo.
  git_cap "$repo" -c core.fsmonitor=false -c core.quotePath=false ls-files -mo --exclude-standard
  if [ "$CAP_RC" -ne 0 ]; then
    GL_STATE=unavailable; GL_BASIS='row-1'
    GL_EXTRA="reason=git-ls-files-error rc=$CAP_RC repo=$repo"
    return 0
  fi
  local lsm=$CAP_OUT

  # Row 6 asks the REPORT repo ($ROOT) for its own branch: a child lane's repo
  # and the fleet root can be different repositories. A non-jj root without a
  # resolvable branch keeps the literal HEAD revision (entering behavior); a
  # `.jj` root without an authoritative mirror is never read from HEAD either —
  # the closure probe is skipped and row 9 still grades the lane mirror.
  local root_rev=''
  if [ "$repo" = "$ROOT" ]; then
    [ -n "$GL_BRANCH" ] && root_rev="refs/heads/$GL_BRANCH"
  elif lane_branch_probe "$ROOT"; then
    root_rev="refs/heads/$BRANCH_OUT"
  fi
  if [ -z "$root_rev" ] && [ ! -d "$ROOT/.jj" ]; then root_rev="HEAD"; fi
  if [ -n "$root_rev" ] && ps_report_closed "$lane" "$LA_ROUND" "$LA_EPOCH" "$root_rev"; then
    GL_STATE=not-open; GL_BASIS='row-6'
    GL_EXTRA="reason=committed-report round=$LA_ROUND path=.ufo/reports/round-$LA_ROUND/$lane.md"
    return 0
  fi

  # -- row 9: authorship -------------------------------------------------------
  # The revision is refs/heads/<branch> when one resolves (a jj commit can leave
  # HEAD one commit ahead of the bookmark, and grading that commit as lane
  # progress would credit work the branch does not carry). A1 fallback: a NON-jj
  # repo with no resolved branch keeps the literal HEAD anchor; a colocated repo
  # never reaches this row unresolved (it exited at the branch gate).
  local rev="HEAD"
  [ -n "$GL_BRANCH" ] && rev="refs/heads/$GL_BRANCH"
  local has_head=0 subj=0 sib=0 sibn=0 n_rowb=0 example='' path mt inwin authors
  local expected
  git_cap "$repo" rev-parse --verify -q "$rev"
  if [ "$CAP_RC" -eq 0 ]; then
    has_head=1
    # Rows A/C keep the entering author line format and parser: this call answers
    # authorship only, so the amendment's no-change rule for rows A/C holds.
    local lg_args=(--since="@$LA_EPOCH" --format='%ct %an <%ae>')
    # Rows A/C scan all post-launch commits on the branch (a subject commit
    # hidden behind a later sibling branch tip still counts). The pathspecs bind
    # the main-repo lanes to the root's own scope; a child lane's repo is the
    # lane scope entire. The revision MUST precede the `--` pathspec.
    local lg_paths=()
    if is_root_lane "$lane"; then
      lg_paths+=(-- . ':(exclude).ufo' ':(exclude,glob)**/*-shokunin/**')
    fi
    git_cap "$repo" log "${lg_args[@]}" "$rev" "${lg_paths[@]+"${lg_paths[@]}"}"
    if [ "$CAP_RC" -ne 0 ]; then
      GL_STATE=unavailable; GL_BASIS='row-1'
      GL_EXTRA="reason=git-log-error rc=$CAP_RC repo=$repo"
      return 0
    fi
    # --since is committer-date based and inclusive at equality (measured on git
    # 2.55.0), so the strict window (launch, NOW] is applied by an explicit %ct
    # comparison here — that is what keeps a commit at ct == launch out of both
    # attribution and closure.
    # F01 (audit-r11 §4 + Appendix B): the old `printf | sed | grep -Fxq`
    # handed a live producer to an early-exit consumer under pipefail. On a long
    # history grep exits at the first match while sed is still writing, sed dies
    # of SIGPIPE (141), pipefail makes the pipeline rc 141, and an exact subject
    # commit that IS present grades progressing-weak (measured: one subject +
    # 12,000 siblings -> other_authored_commits=12001, rc 0). The capture and the
    # membership test are now separate: one awk pass writes the author list into
    # a variable, and a bash pattern match tests WHOLE-LINE membership. No
    # process feeds a reader, so no writer can be killed mid-stream.
    inwin=$(awk -v lo="$LA_EPOCH" -v hi="$NOW_EPOCH" \
      '$1 ~ /^[0-9]+$/ && $1+0 > lo+0 && $1+0 <= hi+0 { print }' <<< "$CAP_OUT")
    if [ -n "$inwin" ]; then
      sibn=$(awk 'END { print NR + 0 }' <<< "$inwin")
      authors=$(awk '{ sub(/^[0-9]+ /, ""); print }' <<< "$inwin")
      expected="$lane-shokunin <$lane@shokunin.local>"
      case $'\n'"$authors"$'\n' in
        *$'\n'"$expected"$'\n'*) subj=1 ;;
        *) sib=1 ;;
      esac
    fi
  fi

  if [ "$subj" -eq 1 ]; then
    GL_STATE=progressing; GL_BASIS='row-A'
    GL_EXTRA='subject_commits=yes window=launch..NOW'
    return 0
  fi

  # Row B: untracked or dirty files newer than the anchor. Only paths git already
  # flags are examined, so a clean tracked file can never be promoted (the
  # spec's named falsifier); ignored trees never enter (`--exclude-standard`).
  if [ -n "$lsm" ]; then
    while IFS= read -r path; do
      [ -n "$path" ] || continue
      in_lane_scope "$lane" "$path" || continue
      mt=$(epoch_mtime "$repo/$path")
      is_int "$mt" || continue
      if [ "$mt" -gt "$LA_EPOCH" ]; then
        n_rowb=$((n_rowb + 1))
        [ -n "$example" ] || example=$path
      fi
    done <<< "$lsm"
  fi
  if [ "$n_rowb" -gt 0 ]; then
    GL_STATE=progressing; GL_BASIS='row-B'
    GL_EXTRA="rowb_files=$n_rowb attribution=inference example=$example"
    return 0
  fi

  if [ "$sib" -eq 1 ]; then
    GL_STATE=progressing-weak; GL_BASIS='row-C'
    GL_EXTRA="other_authored_commits=$sibn"
    return 0
  fi
  GL_STATE=silent; GL_BASIS='row-D'
  # `has_head` keeps its output token (drift keys, drivers); it reports whether
  # the row-9 anchor revision resolved (branch tip, else HEAD in A1 fallback).
  GL_EXTRA="no_subject_commits=yes no_other_commits=yes no_fresh_files=yes has_head=$has_head"
}
